// ========================================== // CONTROLLER LOGIC (UPDATED COMMAND NAMES) // ========================================== // We changed 'A' and 'B' to 'ACT_A' and 'ACT_B' to stop // conflicts with built-in micro:bit button events! const btnCodes = { 'btn-up': 'DIR_U', 'btn-down': 'DIR_D', 'btn-left': 'DIR_L', 'btn-right': 'DIR_R', 'btn-y': 'ACT_Y', 'btn-a': 'ACT_A', 'btn-x': 'ACT_X', 'btn-b': 'ACT_B' }; const buttonStates = {}; const buttons = document.querySelectorAll('.btn'); buttons.forEach(btn => { btn.addEventListener('touchstart', (e) => { e.preventDefault(); if (!buttonStates[btn.id]) { buttonStates[btn.id] = true; btn.classList.add('active'); sendCommand(btnCodes[btn.id] + "_ON"); } }, {passive: false}); btn.addEventListener('touchend', (e) => { e.preventDefault(); if (buttonStates[btn.id]) { buttonStates[btn.id] = false; btn.classList.remove('active'); sendCommand(btnCodes[btn.id] + "_OFF"); } }, {passive: false}); btn.addEventListener('touchcancel', (e) => { if (buttonStates[btn.id]) { buttonStates[btn.id] = false; btn.classList.remove('active'); sendCommand(btnCodes[btn.id] + "_OFF"); } }); });
